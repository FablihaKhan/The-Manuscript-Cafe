CREATE OR REPLACE FUNCTION GetApprovalRate(publisher_id BIGINT) 
RETURNS FLOAT AS $$
DECLARE
    total_submissions INT;
    approved_count INT;
    approval_rate FLOAT;
BEGIN
    SELECT COUNT(*) INTO total_submissions FROM Publish_Request WHERE Publish_Request.publisher_id = GetApprovalRate.publisher_id;
    SELECT COUNT(*) INTO approved_count FROM Approved_books WHERE Approved_books.publisher_id = GetApprovalRate.publisher_id;
    
    IF total_submissions = 0 THEN
        approval_rate := 0;
    ELSE
        approval_rate := (approved_count * 100.0) / total_submissions;
    END IF;
    
    RETURN approval_rate;
END;
$$ LANGUAGE plpgsql;

DROP PROCEDURE financialstatistics();

CREATE OR REPLACE PROCEDURE FinancialStatistics(
    OUT total_revenue DECIMAL(10, 2),
    OUT total_cost DECIMAL(10, 2),
    OUT profit_margin DECIMAL(10, 2)
)
LANGUAGE plpgsql
AS $$
BEGIN
    -- Calculate total revenue
    SELECT SUM(book_price) INTO total_revenue
    FROM Payment;

    -- Calculate total cost
    SELECT SUM(author_gets) + SUM(editor_gets) + SUM(publisher_gets) INTO total_cost
    FROM Payment;

    -- Calculate profit margin
    profit_margin := (total_revenue - total_cost) / total_revenue * 100;
END;
$$;


DO
$$
DECLARE
    revenue DECIMAL(10, 2);
    t_cost DECIMAL(10, 2);
    profit DECIMAL(10, 2);
BEGIN
    CALL FinancialStatistics(revenue,t_cost,profit);
    RAISE NOTICE 'Total Revenue: %', revenue;
    RAISE NOTICE 'Total Cost: %', t_cost;
    RAISE NOTICE 'Profit Margin: %', profit;
END;
$$;



//////////////////////procedure to cascade delete an request
DROP PROCEDURE DeletePublishRequest;
CREATE OR REPLACE PROCEDURE DeletePublishRequest(p_book_name IN VARCHAR) 
AS $$
DECLARE
  v_request_id BIGINT;
BEGIN
  -- Cursor declaration to select multiple request_ids
  FOR v_request_id IN
    SELECT request_id FROM Publish_Requested_books WHERE book_name = p_book_name
  LOOP
    -- Delete associated requests from Publish_Requested_books table
    DELETE FROM Publish_Requested_books WHERE request_id = v_request_id;

    -- Delete requests from Publish_Request table
    DELETE FROM Publish_Request WHERE request_id = v_request_id;

    -- Display success message for each request_id
    RAISE NOTICE 'Request ID % deleted successfully.', v_request_id;
  END LOOP;

  -- Display overall success message
  RAISE NOTICE 'Publish requests and associated records deleted successfully.';
EXCEPTION
  WHEN OTHERS THEN
    -- Display error message
    RAISE NOTICE 'Error: Failed to delete publish requests and associated records.';
END;
$$ LANGUAGE plpgsql;

///////////////////////procedure to send request to all publisher

CREATE OR REPLACE PROCEDURE Send_Publish_Request_To_All_Publishers(
    IN author_id_in BIGINT,
    IN book_name_in VARCHAR(255),
    IN genre_in VARCHAR(50),
    IN pdf_link_in VARCHAR(255)
)
AS $$
DECLARE
    publisher_record Publisher%ROWTYPE;
    request_id_out BIGINT;
    loop_counter INT := 0;
    max_loops INT := 50; -- Number of times to run the loop
BEGIN
    -- Insert into Publish_Requested_books table for each publisher
    FOR publisher_record IN SELECT * FROM Publisher
    LOOP
        EXIT WHEN loop_counter >= max_loops; -- Exit loop after 5 iterations

        -- Generate a new request_id for each iteration
        SELECT nextval('publish_request_request_id_seq') INTO request_id_out;

        -- Insert into Publish_Request table
        INSERT INTO Publish_Request (request_id, author_id, publisher_id)
        VALUES (request_id_out, author_id_in, publisher_record.publisher_id);

        -- Insert into Publish_Requested_books table
        INSERT INTO Publish_Requested_books (request_id, book_name, genre, pdf_link)
        VALUES (request_id_out, book_name_in, genre_in, pdf_link_in);

        loop_counter := loop_counter + 1;
    END LOOP;
END;
$$ LANGUAGE plpgsql;

DROP PROCEDURE SendEditRequest; 

CREATE OR REPLACE PROCEDURE SendEditRequest(
    p_request_id Editor_Books.request_id%TYPE,
    p_approve_id Editor_Books.approve_id%TYPE
)
LANGUAGE plpgsql
AS $$
DECLARE
    editor_rec RECORD;
    editor_cursor CURSOR FOR SELECT editor_id FROM Editor;
BEGIN
    OPEN editor_cursor;
    LOOP
        FETCH NEXT FROM editor_cursor INTO editor_rec;
        EXIT WHEN NOT FOUND;
        INSERT INTO Editor_Books (editor_id, approve_id, request_id, editing_status, edit_start_date,edit_deadline)
        VALUES (editor_rec.editor_id, p_approve_id, p_request_id, 'Not Started', NULL,NULL);
    END LOOP;
    CLOSE editor_cursor;
    
    COMMIT;
END;
$$;

CALL SendEditRequest(p_request_id := 123, p_approve_id := 456);


//////////////////////////trigger on duplicate author

CREATE OR REPLACE FUNCTION prevent_duplicate_author()
RETURNS TRIGGER AS
$$
BEGIN
    IF EXISTS (SELECT 1 FROM Author WHERE email = NEW.email) THEN
        RAISE EXCEPTION 'Author with the same email already exists';
    END IF;
    
    IF EXISTS (SELECT 1 FROM Author WHERE password = NEW.password) THEN
        RAISE EXCEPTION 'Password already exists in the database';
    END IF;

    RETURN NEW;
END;
$$
LANGUAGE plpgsql;

CREATE OR REPLACE TRIGGER prevent_duplicate_author_trigger
BEFORE INSERT ON Author
FOR EACH ROW
EXECUTE FUNCTION prevent_duplicate_author();

//////////////////cannot update password/email that is of another author
CREATE OR REPLACE FUNCTION prevent_duplicate_email_and_password()
RETURNS TRIGGER AS
$$
BEGIN
    IF EXISTS (SELECT 1 FROM Author WHERE email = NEW.email AND id <> NEW.id) THEN
        RAISE EXCEPTION 'Another author with the same email already exists';
    END IF;

    IF EXISTS (SELECT 1 FROM Author WHERE password = NEW.password AND id <> NEW.id) THEN
        RAISE EXCEPTION 'Another author with the same password already exists';
    END IF;

    RETURN NEW;
END;
$$
LANGUAGE plpgsql;

CREATE OR REPLACE TRIGGER prevent_duplicate_email_and_password_trigger
BEFORE UPDATE ON Author
FOR EACH ROW
EXECUTE FUNCTION prevent_duplicate_email_and_password();


/////////////////////trigger for book deletion

CREATE OR REPLACE FUNCTION delete_associated_records()
RETURNS TRIGGER AS $$
BEGIN
    -- Delete records from Payment table associated with the deleted book
    DELETE FROM Payment WHERE book_id = OLD.book_id;

    -- Delete records from Review table associated with the deleted book
    DELETE FROM Review WHERE book_id = OLD.book_id;

    RETURN OLD;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER delete_associated_records_trigger
BEFORE DELETE ON Book
FOR EACH ROW
EXECUTE FUNCTION delete_associated_records();

-- Create a function to calculate the deadline date
CREATE OR REPLACE FUNCTION calculate_edit_deadline()
    RETURNS TRIGGER AS $$
BEGIN
    -- Get the approval date and duration for the approved book
    DECLARE
        approval_date DATE;
        deadline_date DATE;
    BEGIN
        SELECT Approved_books.approval_date
        INTO approval_date
        FROM Approved_books
        WHERE approve_id = NEW.approve_id;

        -- Calculate the deadline date by adding the duration to the approval date
        deadline_date := approval_date + INTERVAL '15 DAY';

        -- Set the calculated deadline date in the Editor_Books table
        NEW.edit_deadline := deadline_date;

        RETURN NEW;
    END;
END;
$$ LANGUAGE plpgsql;


CREATE TRIGGER calculate_edit_deadline_trigger
    BEFORE INSERT ON Editor_Books
    FOR EACH ROW
    EXECUTE FUNCTION calculate_edit_deadline();

FROM Author a
JOIN Book b ON a.id = b.author_id
JOIN Review r ON b.book_id = r.book_id
WHERE a.id IN (
    SELECT author_id
    FROM (
        SELECT author_id, AVG(rating) AS avg_rating
        FROM Author a
        JOIN Book b ON a.id = b.author_id
        JOIN Review r ON b.book_id = r.book_id
        GROUP BY author_id
        ORDER BY avg_rating DESC
        LIMIT 3
    ) AS top_avg_ratings
)
